Establish the RMAN monitoring baseline

Using your existing DGDB Oracle container, independently investigate and produce a read-only report identifying:
- Recent RMAN backup jobs and their statuses.
- Most recent successful database and archived redo log backups.
- Archived redo log sequences represented in backup metadata.


##################################################################
# connect as SYSDBA to the CDB root and run:
# This view covers backup jobs;
# STATUS distinguishes COMPLETED, COMPLETED WITH WARNINGS, COMPLETED WITH ERRORS, and FAILED
##################################################################
SET LINESIZE 180
SET PAGESIZE 100

COLUMN started FORMAT A19
COLUMN ended   FORMAT A19
COLUMN status  FORMAT A23
COLUMN input_type FORMAT A15

SELECT session_key,
       TO_CHAR(start_time, 'YYYY-MM-DD HH24:MI:SS') AS started,
       TO_CHAR(end_time,   'YYYY-MM-DD HH24:MI:SS') AS ended,
       input_type,
       status,
       output_device_type AS device,
       output_bytes_display AS output_size,
       time_taken_display AS duration
FROM v$rman_backup_job_details
ORDER BY start_time DESC
FETCH FIRST 20 ROWS ONLY;

       b.completion_time,
       r.thread#,
       r.sequence#,

FROM v$backup_redolog r
JOIN v$backup_set b
  ON b.set_stamp = r.set_stamp
 AND b.set_count = r.set_count

# starting with one query that lists the unbacked sequences, including their completion timestamps
WITH archived_logs AS (
    SELECT thread#,
           sequence#,
           completion_time,
           backup_count
    FROM v$archived_log
    WHERE sequence# > 14 -- this avoids counting the duplicate copy of sequence 1, but im not sure how practical this is 
)
SELECT *
FROM archived_logs
WHERE backup_count = 0;




WITH rman_sequences AS (
    SELECT b.recid AS backup_set_key,
        b.completion_time,
        r.thread#,
        r.sequence#,
        r.first_change#,
        r.next_change#
    FROM v$backup_redolog r
    JOIN v$backup_set b
    ON b.set_stamp = r.set_stamp
    AND b.set_count = r.set_count
    ORDER BY b.completion_time DESC, r.thread#, r.sequence#
)
SELECT *
FROM rman_sequences;
